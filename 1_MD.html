import os import zipfile if not os.path.exists('Held_out_set_extracted'): with zipfile.ZipFile('Held-out_set.zip', 'r') as zip_ref: zip_ref.extractall('Held_out_set_extracted') base_dir = 'Held_out_set_extracted' subfolders = os.listdir(base_dir) dataset_path = os.path.join(base_dir, subfolders[0] if len(subfolders) == 1 else '') all_files = sorted([f for f in os.listdir(dataset_path) if f.endswith('.wav')]) print(f'Total audio files found: {len(all_files)}') excluded_files = [] valid_files = [] for f in all_files: f_path = os.path.join(dataset_path, f) if os.path.getsize(f_path) > 1000: valid_files.append(f) else: excluded_files.append({'file': f, 'reason': 'File size too small or corrupted header'}) print(f'Valid files passing quality check: {len(valid_files)}') print(f'Excluded files list: {excluded_files}')

Task 1.3: Channel Voting & Final Consensus

Since each physical experiment was recorded simultaneously across four independent microphone channels (M1–M4), we aggregated predictions via majority voting to determine the unified final consensus per experiment.

def extract_experiment_id(filename): parts = filename.replace('.wav', '').split('_') return parts[-1] if len(parts) > 1 else filename df_results['experiment'] = df_results['file'].apply(extract_experiment_id) def mode_vote(series): mode_val = series.mode() return mode_val.iloc[0] if not mode_val.empty else series.iloc[0] final_predictions = df_results.groupby('experiment').agg({ 'instrument': mode_vote, 'tuning': mode_vote, 'distance_cm': 'mean', 'conclusion': mode_vote }).reset_index() final_predictions.to_csv('Held_out_set_final_consensus.csv', index=False)
